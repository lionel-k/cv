\cvsection{6. Audits et évaluations techniques}

\begin{cvgroup}
\cvsubsection{Audit 1 : Évaluation du système NEMIDIS POS}

\begin{cvmetadata}
\textbf{Période :} 2025 \\
\textbf{Client :} Nemidis Technologies / Office Burundais des Recettes (OBR) \\
\textbf{Rôle :} CTO / Architecte logiciel — auditeur technique
\end{cvmetadata}

\begin{cvprojectsummary}
Évaluation approfondie d'une plateforme critique de facturation électronique soumise aux obligations fiscales de l'OBR, avec focus sur l'architecture existante, le respect des normes de développement et les axes d'amélioration.
\end{cvprojectsummary}

\begin{cvdetailslist}{Travaux réalisés}
  \item {Évaluation du code source, de l'architecture applicative et de la conformité aux bonnes pratiques.}
  \item {Cartographie des composants, bases de données PostgreSQL et services d'arrière-plan.}
  \item {Analyse des intégrations avec les services EBMS de l'OBR et documentation des flux métier (facturation, stock, synchronisation fiscale).}
  \item {Évaluation des mécanismes de sécurité, permissions et audit trail.}
  \item {Production d'une documentation d'architecture de référence pour développeurs et parties prenantes institutionnelles.}
  \item {Identification des risques et recommandations techniques pour les évolutions ASYCUDA–EBMS et du POS.}
\end{cvdetailslist}

\cvdetailblock{Résultats}{Documentation de référence du SI · Réduction du risque de dépendance à la connaissance individuelle · Base pour l'intégration ASYCUDA–EBMS–POS-OBR et les évolutions réglementaires.}

\cvdetailblock{Compétences démontrées}{Évaluation de SI · Audit applicatif · Analyse d'architecture · Normes et standards · Documentation technique · Gouvernance applicative · Recommandations d'amélioration}
\end{cvgroup}

\begin{cvgroup}
\cvsubsection{Audit 2 : Évaluation de l'architecture de gestion des accès (Abbove)}

\begin{cvmetadata}
\textbf{Période :} Mai 2026 – Juin 2026 \\
\textbf{Client :} Abbove (plateforme patrimoniale — institutions financières) \\
\textbf{Rôle :} Ingénieur logiciel senior / Auditeur technique
\end{cvmetadata}

\begin{cvprojectsummary}
Évaluation du système de gestion des accès et des droits utilisateurs : documentation des mécanismes d'autorisation, analyse des risques opérationnels et recommandations de gouvernance conformes aux exigences de sécurité.
\end{cvprojectsummary}

\begin{cvdetailslist}{Travaux réalisés}
  \item {Cartographie complète du modèle de contrôle d'accès (RBAC, ACL, ABAC) sur stack GraphQL / microservices.}
  \item {Évaluation des processus de propagation et de révocation des droits utilisateurs.}
  \item {Identification des risques liés aux accès hérités et aux autorisations persistantes.}
  \item {Documentation de l'architecture de sécurité et présentation exécutive aux parties prenantes.}
  \item {Proposition d'un mécanisme automatisé de gouvernance et de nettoyage des habilitations.}
\end{cvdetailslist}

\cvdetailblock{Technologies}{Redis · PostgreSQL · Docker · GraphQL · Ruby on Rails · Microservices · API REST (JSON)}

\cvdetailblock{Résultats}{Mise en évidence des risques de conservation involontaire des accès · Plan d'amélioration de la gouvernance des habilitations · Documentation de référence pour équipes métier et techniques.}

\cvdetailblock{Compétences démontrées}{Évaluation de SI · Audit de sécurité · Analyse des risques · Gouvernance des accès · Documentation technique · Présentation exécutive · Conformité opérationnelle}
\end{cvgroup}
