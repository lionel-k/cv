\cvsubsection{Audit 1 : Audit d'architecture du système NEMIDIS POS}

\begin{cvparagraph}
\textbf{Période :} 2025 \\
\textbf{Client :} Nemidis Technologies \\
\textbf{Rôle :} CTO / Architecte logiciel
\end{cvparagraph}

\begin{cvprojectsummary}
Prise en charge d'une plateforme critique de facturation électronique utilisée dans le cadre des obligations fiscales de l'Office Burundais des Recettes (OBR).
\end{cvprojectsummary}

\cvdetailslabel{Travaux réalisés}
\begin{cvdetailitems}
  \item {Analyse du code source et de l'architecture existante.}
  \item {Cartographie des composants applicatifs, bases de données et services d'arrière-plan.}
  \item {Analyse des intégrations avec les services EBMS de l'OBR.}
  \item {Documentation des flux métier de facturation, gestion de stock, synchronisation fiscale et géolocalisation.}
  \item {Évaluation des mécanismes de sécurité, permissions et audit trail.}
  \item {Production d'une documentation d'architecture destinée aux développeurs et aux parties prenantes.}
  \item {Identification des axes d'amélioration pour les futures évolutions du système.}
\end{cvdetailitems}

\begin{cvparagraph}
\textbf{Résultats :} Constitution d'une documentation de référence du système · Réduction du risque de dépendance à la connaissance individuelle · Facilitation de l'intégration des nouveaux développeurs · Base utilisée pour les futurs projets d'intégration ASYCUDA–EBMS et les évolutions du POS.
\end{cvparagraph}

\begin{cvparagraph}
\textit{Compétences démontrées :} Audit applicatif · Cartographie de système d'information · Analyse d'architecture · Documentation technique · Gouvernance applicative · Intégration de systèmes
\end{cvparagraph}

\cvsubsection{Audit 2 : Audit de l'architecture de gestion des accès (Abbove)}

\begin{cvparagraph}
\textbf{Période :} Mai 2026 – Juin 2026 \\
\textbf{Client :} Abbove (plateforme de gestion patrimoniale utilisée par des institutions financières) \\
\textbf{Rôle :} Ingénieur logiciel senior / Auditeur technique
\end{cvparagraph}

\begin{cvprojectsummary}
Analyse du système de gestion des accès et des droits utilisateurs afin de documenter les mécanismes d'autorisation, identifier les risques opérationnels et proposer des améliorations de gouvernance.
\end{cvprojectsummary}

\cvdetailslabel{Travaux réalisés}
\begin{cvdetailitems}
  \item {Cartographie complète du modèle de contrôle d'accès de la plateforme.}
  \item {Analyse des mécanismes RBAC, ACL, ABAC et des règles métier associées.}
  \item {Évaluation des processus de propagation et de révocation des droits utilisateurs.}
  \item {Identification des risques liés aux accès hérités et aux autorisations persistantes.}
  \item {Documentation de l'architecture de sécurité sous forme de schémas et présentations exécutives.}
  \item {Présentation des résultats à des parties prenantes techniques et non techniques.}
  \item {Proposition d'un mécanisme automatisé de nettoyage et de gouvernance des droits d'accès.}
\end{cvdetailitems}

\begin{cvparagraph}
\textbf{Résultats :} Mise en évidence des risques de conservation involontaire des accès · Définition d'un plan d'amélioration de la gouvernance des habilitations · Production d'une documentation de référence utilisée par les équipes métier et techniques.
\end{cvparagraph}

\begin{cvparagraph}
\textit{Compétences démontrées :} Audit de système d'information · Gouvernance des accès · Architecture logicielle · Analyse des risques · Documentation technique · Présentation exécutive · Sécurité applicative
\end{cvparagraph}
